func (fg *FactGuard) Verify(ctx context.Context, gen *LLMOutput) (*Resume, error) {
  // 1. Enforce strict bounds against candidate master resume
  if err := fg.Validator.AssertBounds(fg.Master, gen); err != nil {
    return nil, fmt.Errorf("fact-guard failure: %w", err)
  }

  // 2. Keep contact & education entirely outside LLM reach
  gen.Contact = fg.Master.DeterministicContact
  gen.Education = fg.Master.DeterministicEducation

  // 3. Cost-optimized LaTeX rendering (~87% token cost cut)
  return fg.RenderLaTeX(ctx, gen)
}

func (p *Proxy) Execute(ctx context.Context, tx *Tx) (*Receipt, error) {
  // Atomic Redis Mutex with strict 10s TTL lease
  lock, err := p.RedisLock.Acquire(ctx, tx.IdempKey)
  if err != nil {
    return nil, ErrConcurrencyCollision
  }
  defer lock.Release(ctx)

  // Config-driven route across 18 banking issuers
  handler := p.Registry.ResolveIssuer(tx.IssuerCode)
  return handler.RedeemOrAccrue(ctx, tx)
}
